cmake_minimum_required(VERSION 3.20)
project(device-driver-binding-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This application requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(DDBE_ANALYZE "Enable MSVC static analysis" OFF)
option(DDBE_LIVE_TESTS "Enable read-only host integration tests" ON)
add_library(ddbe_core src/core.cpp src/windows.cpp)
target_include_directories(ddbe_core PUBLIC src)
target_compile_definitions(ddbe_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(ddbe_core PUBLIC setupapi cfgmgr32 advapi32)
add_executable(device-driver-binding-explainer src/main.cpp)
enable_language(RC)
target_sources(device-driver-binding-explainer PRIVATE src/version.rc)
target_link_libraries(device-driver-binding-explainer PRIVATE ddbe_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(ddbe_tests tests/tests.cpp)
  target_link_libraries(ddbe_tests PRIVATE ddbe_core)
  add_test(NAME fixtures COMMAND ddbe_tests)
  add_test(NAME help COMMAND device-driver-binding-explainer --help)
  add_test(NAME version COMMAND device-driver-binding-explainer --version)
  add_test(NAME cli_contract COMMAND ${CMAKE_COMMAND}
    "-DAPP=$<TARGET_FILE:device-driver-binding-explainer>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  if(DDBE_LIVE_TESTS)
    add_test(NAME live_read_only COMMAND ddbe_tests --live)
    add_test(NAME live_cli_json COMMAND ${CMAKE_COMMAND}
      "-DAPP=$<TARGET_FILE:device-driver-binding-explainer>" -DLIVE=ON -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  endif()
endif()
foreach(target ddbe_core device-driver-binding-explainer ddbe_tests)
  if(TARGET ${target} AND MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
    if(DDBE_ANALYZE)
      target_compile_options(${target} PRIVATE /analyze)
    endif()
  endif()
endforeach()
install(TARGETS device-driver-binding-explainer RUNTIME DESTINATION .)
install(FILES README.md LICENSE VALIDATION.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "device-driver-binding-explainer-${PROJECT_VERSION}-windows-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
